# 자료구조 queue 구현
class CircleQueue:
  def __init__(self,size):
    self.size=size
    self.arr = [None]*size
    self.rear =0
    self.front =0
  def isFull(self):
    return self.front == (self.rear+1)%self.size
  def isEmpty(self):
    return self.front == self.rear
  def enqueue(self,data):
    if self.isFull():
      print("꽉참")
      return
    self.rear =((self.rear+1)%self.size)
    self.arr[self.rear] = data
    return self.arr
  def dequeue(self):
    if self.isEmpty():
      print("공백")
      return
    self.front = ((self.front + 1)%self.size)
    dedata = self.arr[self.front]
    self.arr[self.front] = None
    return dedata

# Py programming
class oneTwo: #클래스 함수 - 메서드 
  def __init__(self, name, score): # 메서드중 init은 생성자⭐️ 안에 변수 - 필드, 속성
    self.name = name
    self.score = score
  def get_score(self): #
    print(f'{self.name}의 합계는 {sum(self.score)}입니다')
  def get_AVG(self):
    print(f'{self.name}의 합계는 {sum(self.score)/len(self.score)}')
s1 = oneTwo('감강현',[100,100,100])
s1.get_score()
s1.get_AVG()
