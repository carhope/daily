# 자료구조 queue 구현
class CircleQueue:
  def __init__(self,size):
    self.size=size
    self.arr = [None]*size
    self.rear =0
    self.front =0
  def isFull(self):
    return self.front == (self.rear+1)%self.size
  def isEmpty(self):
    return self.front == self.rear
  def enqueue(self,data):
    if self.isFull():
      print("꽉참")
      return
    self.rear =((self.rear+1)%self.size)
    self.arr[self.rear] = data
    return self.arr
  def dequeue(self):
    if self.isEmpty():
      print("공백")
      return
    self.front = ((self.front + 1)%self.size)
    dedata = self.arr[self.front]
    self.arr[self.front] = None
    return dedata
