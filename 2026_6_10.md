# 개발자 환경구축 프로젝트 beatkim DB 작성 완료
create database beatkim_db;

use beatkim_db;
create table user(
	user_id int primary key,
    user_name varchar(100),
    user_pw varchar(100)
);

create table singer(
	singer_id int primary key,
    singer_name varchar(100)
);

create table song(
	song_id int primary key,
    song_name varchar(255),
    song_explain varchar(255),
    singer_id int,
    
    foreign key(singer_id)
    references singer(singer_id)
);

create table likelist(
	user_id int ,
    song_id int ,
    
    primary key(user_id,song_id),
    
    foreign key(user_id)
    references user(user_id),
    
    foreign key(song_id)
    references song(song_id)
);
    
이때 likelist의 primary key는 두개임.
why : 찜 목록에 노래가 여러개 들어갈 수 있기 떄문. primary키 로 지정하지 않으면 한개만 들어감.
